cmake_minimum_required(VERSION 3.29)
project(ysm LANGUAGES CXX)

if (NOT YSM_DEPENDENCY_PROVIDER STREQUAL "conan")
    message(FATAL_ERROR "Please build the project using bootstrap scripts.")
endif ()

option(YSM_BUILD_BENCHMARKS "Build the ysm microbenchmark executable" OFF)
option(YSM_ENABLE_DEBUG_LOG "Compile Debug and Trace logging into native targets" OFF)
option(YSM_ENABLE_TRACY "Enable Tracy CPU profiling instrumentation" OFF)

include(CTest)
include(third-party/import.cmake)

add_subdirectory(modules/proto)
add_subdirectory(modules/core)
add_subdirectory(modules/gfx)
add_subdirectory(modules/codec)
add_subdirectory(modules/java)
add_subdirectory(modules/legacy)
add_subdirectory(modules/lib-jni)
if (YSM_BUILD_BENCHMARKS)
    add_subdirectory(modules/benchmark)
endif ()
if (BUILD_TESTING)
    add_subdirectory(modules/test)
endif ()
